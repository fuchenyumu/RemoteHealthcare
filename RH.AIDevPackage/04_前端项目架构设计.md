# 前端架构设计方案（基于实际PureAdmin项目）

## 1. 项目概述

基于Vue3 + PureAdmin + Element Plus构建的区域妇幼保健远程诊疗平台前端，采用现代化前端技术栈，具备完整的权限管理、响应式设计和国际化支持。

```typescript
  ["./modules/**/*.ts", "!./modules/**/remaining.ts"],
  { eager: true }

// 路由层级处理

  formatFlatteningRoutes(buildHierarchyTree(ascending(routes.flat(Infinity))))
);

// 路由守卫（完整权限控制）
router.beforeEach((to: ToRouteType, _from, next) => {
  NProgress.start();
  
  const userInfo = storageLocal().getItem<DataInfo<number>>(userKey);
  
  // 权限检查
  if (to.meta?.roles && !isOneOfArray(to.meta?.roles, userInfo?.roles)) {
    next({ path: "/error/403" });
    return;
  }
  
  // Token检查和刷新
  if (userInfo && !isAllEmpty(usePermissionStoreHook().wholeMenus)) {
    // 正常路由处理
  } else if (to.path !== "/login") {
    // 重定向到登录页
    removeToken();
    next({ path: "/login" });
    return;
  }
  
  next();
});
```

### 3.2 权限控制组件
```vue
<!-- 基于PureAdmin的权限组件实现 -->
<script setup lang="ts">
import { ref, computed } from "vue";
import { useUserStoreHook } from "@/store/modules/user";

interface Props {
  value: string | Array<string>;  // 权限标识
  default?: string | Array<string>;
  fallback?: string;
}

const props = withDefaults(defineProps<Props>(), {
  default: "",
  fallback: "/error/403"
});

const userStore = useUserStoreHook();
const visible = ref(false);

// 权限检查逻辑
const checkAuth = () => {
  const { value, default: defaultVal } = props;
  const userRoles = userStore.roles;
  
  // 超级管理员一直显示
  if (userRoles.includes("admin")) {
    visible.value = true;
    return;
  }

  if (Array.isArray(value)) {
    visible.value = value.some(auth => userRoles.includes(auth));
  } else if (typeof value === "string") {
    visible.value = value === defaultVal ? true : userRoles.includes(value);
  }
};

checkAuth();
</script>

<template>
  <span v-if="visible">
    <slot />
  </span>
</template>
```

### 4.1 用户状态管理（基于实际PureAdmin）
```typescript
// store/modules/user.ts
import { defineStore } from "pinia";
import { store } from "../utils";
import { getLogin, refreshTokenApi } from "@/api/user";
import { useMultiTagsStoreHook } from "./multiTags";
import { setToken, removeToken, userKey } from "@/utils/auth";
import { router, resetRouter, routerArrays } from "../utils";

export const useUserStore = defineStore("pure-user", {
  state: (): userType => ({
    avatar: storageLocal().getItem<DataInfo<number>>(userKey)?.avatar ?? "",
    username: storageLocal().getItem<DataInfo<number>>(userKey)?.username ?? "",
    nickname: storageLocal().getItem<DataInfo<number>>(userKey)?.nickname ?? "",
    roles: storageLocal().getItem<DataInfo<number>>(userKey)?.roles ?? [],
    permissions: storageLocal().getItem<DataInfo<number>>(userKey)?.permissions ?? [],
    verifyCode: "",
    currentPage: 0,
    isRemembered: false,
    loginDay: 7
  }),
  actions: {
    // 登录
    async loginByUsername(data) {
      return new Promise<UserResult>((resolve, reject) => {
        getLogin(data)
          .then(data => {
            if (data?.success) setToken(data.data);
            resolve(data);
          })
          .catch(error => reject(error));
      });
    },
    
    // 登出
    logOut() {
      this.username = "";
      this.roles = [];
      this.permissions = [];
      removeToken();
      useMultiTagsStoreHook().handleTags("equal", [...routerArrays]);
      resetRouter();
      router.push("/login");
    },
    
    // 刷新token
    async handRefreshToken(data) {
      return new Promise<RefreshTokenResult>((resolve, reject) => {
        refreshTokenApi(data)
          .then(data => {
            if (data) {
              setToken(data.data);
              resolve(data);
            }
          })
          .catch(error => reject(error));
      });
    }
  }
});

export function useUserStoreHook() {
  return useUserStore(store);
}
```

```typescript
// store/modules/consult.ts（待开发）
import { defineStore } from "pinia";
import { ref } from "vue";
import type { ConsultApply, ConsultDetail } from "@/types/consult";
import * as consultAPI from "@/api/consult";

export const useConsultStore = defineStore("consult", () => {
  // 状态
  const consultList = ref<ConsultApply[]>([]);
  const currentConsult = ref<ConsultDetail | null>(null);
  const loading = ref(false);
  
  // 获取会诊列表
  const fetchConsultList = async (params: {
    page?: number;
    size?: number;
    status?: number;
  }) => {
    try {
      loading.value = true;
      const response = await consultAPI.getConsultList(params);
      consultList.value = response.data.records;
      return response;
    } catch (error) {
      throw error;
    } finally {
      loading.value = false;
    }
  };
  
  // 获取会诊详情
  const fetchConsultDetail = async (applyId: string) => {
    try {
      loading.value = true;
      const response = await consultAPI.getConsultDetail(applyId);
      currentConsult.value = response.data.consultApply;
      return response.data;
    } catch (error) {
      throw error;
    } finally {
      loading.value = false;
    }
  };
  
  // 创建会诊申请
  const createConsult = async (consultData: CreateConsultRequest) => {
    try {
      loading.value = true;
      const response = await consultAPI.createConsult(consultData);
      consultList.value.unshift(response.data);
      return response.data;
    } catch (error) {
      throw error;
    } finally {
      loading.value = false;
    }
  };
  
  return {
    consultList,
    currentConsult,
    loading,
    fetchConsultList,
    fetchConsultDetail,
    createConsult
  };
});
```

## 5. API接口封装

### 5.1 HTTP请求封装（基于实际PureHttp）
```typescript
// utils/http/index.ts
import Axios from "axios";
import { getToken, formatToken } from "@/utils/auth";
import { useUserStoreHook } from "@/store/modules/user";

class PureHttp {
  private static requests = [];
  private static isRefreshing = false;
  private static axiosInstance = Axios.create({
    baseURL: import.meta.env.VITE_APP_BACKEND_URL || "http://127.0.0.1:8828",
    timeout: 10000,
    headers: {
      Accept: "application/json, text/plain, */*",
      "Content-Type": "application/json"
    }
  });

  // 请求拦截器
  private httpInterceptorsRequest(): void {
    PureHttp.axiosInstance.interceptors.request.use(
      async (config) => {
        const whiteList = ["/refresh-token", "/login"];
        
        // 白名单接口直接通过
        if (whiteList.some(url => config.url.endsWith(url))) {
          return config;
        }
        
        // Token管理
        const data = getToken();
        if (data) {
          const now = new Date().getTime();
          const expired = parseInt(data.expires) - now <= 0;
          
          if (expired) {
            // Token过期处理
            if (!PureHttp.isRefreshing) {
              PureHttp.isRefreshing = true;
              try {
                const res = await useUserStoreHook().handRefreshToken({
                  refreshToken: data.refreshToken
                });
                const token = res.data.accessToken;
                config.headers["Authorization"] = formatToken(token);
                PureHttp.requests.forEach(cb => cb(token));
                PureHttp.requests = [];
              } finally {
                PureHttp.isRefreshing = false;
              }
            }
            return new Promise(resolve => {
              PureHttp.requests.push((token: string) => {
                config.headers["Authorization"] = formatToken(token);
                resolve(config);
              });
            });
          } else {
            config.headers["Authorization"] = formatToken(data.accessToken);
          }
        }
        
        return config;
      },
      error => Promise.reject(error)
    );
  }

  // 响应拦截器
  private httpInterceptorsResponse(): void {
    PureHttp.axiosInstance.interceptors.response.use(
      (response) => response.data,
      (error) => {
        const $error = error;
        $error.isCancelRequest = Axios.isCancel($error);
        return Promise.reject($error);
      }
    );
  }

  // 通用请求方法
  public request<T>(
    method: RequestMethods,
    url: string,
    param?: AxiosRequestConfig,
    config?: PureHttpRequestConfig
  ): Promise<T> {
    const requestConfig = { method, url, ...param, ...config };
    
    return new Promise((resolve, reject) => {
      PureHttp.axiosInstance
        .request(requestConfig)
        .then(resolve)
        .catch(reject);
    });
  }

  // GET请求
  public get<T, P>(url: string, params?: AxiosRequestConfig<P>, config?: PureHttpRequestConfig): Promise<T> {
    return this.request<T>("get", url, { params }, config);
  }

  // POST请求
  public post<T, P>(url: string, params?: AxiosRequestConfig<P>, config?: PureHttpRequestConfig): Promise<T> {
    return this.request<T>("post", url, params, config);
  }
}

export const http = new PureHttp();
```

```typescript
const modulePrefix = "/api/rc/consultations";
```

## 9. 与后端 RH.WebApi 的衔接

- **基础 BASE_URL**：通过 `src/config/platform-config` 读取，默认为 `/api`
- **接口约定**：
  - 会诊主表：`GET/POST /api/rc/consultations`
  - 审核：`PUT /api/rc/consultations/{id}/audit`
  - 排期：`PUT /api/rc/consultations/{id}/schedule`
  - 报告签署：`POST /api/rc/reports`、`PUT /api/rc/reports/{id}/sign`
  - 附件上传：复用系统上传接口 `/api/upload`
- **权限编码映射**：

| 前端路由 | 按钮权限值 | 后端功能编码 | 字典依赖 |
| --- | --- | --- | --- |
| `consult/list` | `rc:consult:view` | `purest_function.code=RC_CONSULT_VIEW` | `RC_CONSULT_STATUS` |
| `consult/apply` | `rc:consult:add` | `RC_CONSULT_CREATE` | `RC_PATIENT_TYPE`, `RC_EMERGENCY_LEVEL` |
| `consult/review` | `rc:consult:audit` | `RC_CONSULT_AUDIT` | `RC_TIMELINE_EVENT` |
| `rtc-demo` | `rc:consult:rtc` | `RC_CONSULT_RTC` | `RC_MEMBER_ROLE` |
| `patient/pack` | `rc:patient:pack` | `RC_PATIENT_PACK` | `RC_PACK_STATUS`, `RC_SOURCE_SYSTEM` |

- **字典获取**：统一通过 `useDictStore()` 批量拉取：

```ts
const dictStore = useDictStore();
await dictStore.init(["RC_PATIENT_TYPE", "RC_CONSULT_STATUS", "RC_EMERGENCY_LEVEL"]);
const consultStatusOptions = dictStore.get("RC_CONSULT_STATUS");
```

- **TraceId透传**：HTTP 拦截器在响应头中读取 `x-trace-id`，写入控制台日志，便于与后端 Log 关联。

- **错误码规范**：后端采用 `{ success, data, errorCode, errorMessage }`，前端需对 `errorCode` 做统一映射（如401重定向登录、403提示无权限）。
