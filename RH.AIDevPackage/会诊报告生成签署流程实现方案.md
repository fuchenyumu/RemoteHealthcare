# 会诊报告生成和签署流程实现方案

## 一、需求分析

基于现有项目架构，需要实现以下核心功能：

### 1.1 业务流程
```
会诊结束 → 生成报告草稿 → 专家填写 → 电子签名 → 报告归档 → 同步EMR
```

### 1.2 核心功能点
- **报告生成**：基于会诊记录自动生成报告模板
- **报告编辑**：支持多格式内容编辑（诊断、建议、治疗方案）
- **电子签名**：支持手写签名、CA证书签名
- **签署流程**：支持多专家会签、顺序签署
- **报告归档**：签署完成后自动归档
- **数据同步**：同步至院内EMR系统

```typescript
// api/rc/report.ts

const modulePrefix = "/api/rc/reports";

// 生成报告
export const generateReport = (consultId: string) => {
  return http.post<ReportDetail>(`${modulePrefix}/generate`, { consultId });
};

// 保存草稿
export const saveDraft = (data: {
  reportId: string;
  summary: string;
  diagnosis: string;
  treatmentAdvice: string;
  followUpPlan: string;
}) => {
  return http.put<ReportDetail>(`${modulePrefix}/draft`, data);
};

// 提交签署
export const submitForSign = (reportId: string) => {
  return http.post<ReportDetail>(`${modulePrefix}/${reportId}/submit`, {});
};

// 电子签名
export const signReport = (data: {
  reportId: string;
  signatureData: string;  // Base64签名数据
  signMethod: 'HANDWRITE' | 'CA' | 'BIO';
}) => {
  return http.post<ReportDetail>(`${modulePrefix}/${data.reportId}/sign`, data);
};

// 获取签署状态
export const getSignStatus = (reportId: string) => {
  return http.get<SignFlow[]>(`${modulePrefix}/${reportId}/sign-flow`);
};

// 归档报告
export const archiveReport = (reportId: string) => {
  return http.post<ReportDetail>(`${modulePrefix}/${reportId}/archive`, {});
};
```

```vue
<script setup lang="ts">
import { ref, computed, watch } from "vue";
import { ElMessage } from "element-plus";

const props = defineProps<{
  modelValue: boolean;
  reportId: string;
}>();

const emit = defineEmits<{
  "update:modelValue": [value: boolean];
  signed: [data: { signatureData: string; signMethod: string }];
}>();

const visible = computed({
  get: () => props.modelValue,
  set: (val) => emit("update:modelValue", val)
});

const signMode = ref<"HANDWRITE" | "CA" | "BIO">("HANDWRITE");
const caPin = ref("");
const bioLoading = ref(false);
const signing = ref(false);

// 手写签名相关
const canvas = ref<HTMLCanvasElement>();
const ctx = ref<CanvasRenderingContext2D | null>();
const isDrawing = ref(false);
const paths = ref<any[]>([]);

// 监听模式切换，重置状态
watch(signMode, () => {
  caPin.value = "";
  bioLoading.value = false;
});

// 手写签名 - 开始绘制
const startDrawing = (e: MouseEvent | TouchEvent) => {
  if (signMode.value !== "HANDWRITE") return;
  
  isDrawing.value = true;
  const pos = getMousePos(e);
  
  ctx.value?.beginPath();
  ctx.value?.moveTo(pos.x, pos.y);
  
  // 记录路径用于撤销
  paths.value.push({ type: "start", x: pos.x, y: pos.y });
};

// 手写签名 - 绘制中
const draw = (e: MouseEvent | TouchEvent) => {
  if (!isDrawing.value || signMode.value !== "HANDWRITE") return;
  
  const pos = getMousePos(e);
  ctx.value?.lineTo(pos.x, pos.y);
  ctx.value?.stroke();
  
  paths.value.push({ type: "draw", x: pos.x, y: pos.y });
};

// 手写签名 - 停止绘制
const stopDrawing = () => {
  if (!isDrawing.value) return;
  isDrawing.value = false;
  ctx.value?.closePath();
};

// 获取鼠标位置
const getMousePos = (e: MouseEvent | TouchEvent) => {
  const rect = canvas.value?.getBoundingClientRect();
  const clientX = "touches" in e ? e.touches[0].clientX : e.clientX;
  const clientY = "touches" in e ? e.touches[0].clientY : e.clientY;
  
  return {
    x: clientX - (rect?.left || 0),
    y: clientY - (rect?.top || 0)
  };
};

// 清空画布
const clearCanvas = () => {
  if (!ctx.value || !canvas.value) return;
  ctx.value.clearRect(0, 0, canvas.value.width, canvas.value.height);
  paths.value = [];
};

// 撤销
const undo = () => {
  if (!ctx.value || !canvas.value || paths.value.length === 0) return;
  
  // 重新绘制所有路径，除了最后一条
  ctx.value.clearRect(0, 0, canvas.value.width, canvas.value.height);
  const newPaths: any[] = [];
  
  paths.value.forEach((path, index) => {
    if (index === paths.value.length - 1) return;
    
    if (path.type === "start") {
      ctx.value?.beginPath();
      ctx.value?.moveTo(path.x, path.y);
```

```csharp
            var members = await _db.Queryable<RcConsultationMemberEntity>()

        public async Task<ReportDto> SignReportAsync(SignReportCommand command)
        {
            // 1. 验证签署权限
            var flow = await _db.Queryable<RcReportSignFlowEntity>()
                .Where(f => f.ReportId == command.ReportId && f.SignerId == CurrentUser.Id)

            if (flow.SignStatus != "PENDING")
                throw new UserFriendlyException("当前状态不允许签署");

            // 2. 验证签署顺序
            var previousFlows = await _db.Queryable<RcReportSignFlowEntity>()
                .Where(f => f.ReportId == command.ReportId && f.SignOrder < flow.SignOrder)
                .ToListAsync();

            if (previousFlows.Any(f => f.SignStatus != "SIGNED"))
                throw new UserFriendlyException("请等待前序签署人完成签署");

            // 3. 生成签名文件
            var signatureFileId = await SaveSignatureFile(command);

            // 4. 更新签署记录
            flow.SignStatus = "SIGNED";
            flow.SignTime = DateTime.Now;
            flow.SignatureFileId = signatureFileId;
            flow.SignIp = command.IpAddress;
            flow.SignDevice = command.UserAgent;
            flow.UpdateBy = CurrentUser.Id;
            flow.UpdateTime = DateTime.Now;

            var allFlows = await _db.Queryable<RcReportSignFlowEntity>()
                .Where(f => f.ReportId == command.ReportId)
                .ToListAsync();

            if (allFlows.All(f => f.SignStatus == "SIGNED"))
            {
                // 所有签署完成，触发归档
                await CompleteReportSigning(command.ReportId);

            await _eventBus.PublishAsync(new ReportSignedEvent
            {
                ReportId = command.ReportId,
                SignerId = CurrentUser.Id,
                SignFlowId = flow.Id
            });

                .Where(r => r.Id == command.ReportId)
                .FirstAsync();
```

### 6.2 后台同步任务

```csharp
// PurestAdmin.BackgroundService/Jobs/ReportSyncJob.cs
using System;
using System.Threading.Tasks;
using PurestAdmin.Core;
using PurestAdmin.SqlSugar.Entities;

namespace PurestAdmin.BackgroundService.Jobs
{
    public class ReportSyncJob
    {
        private readonly ISqlSugarClient _db;
        private readonly IEMRIntegrationService _emrService;

        public ReportSyncJob(ISqlSugarClient db, IEMRIntegrationService emrService)
        {
            _db = db;
            _emrService = emrService;
        }

        public async Task SyncToEMR(Guid reportId)
        {
            var task = await _db.Queryable<RcSyncTaskEntity>()
                .Where(t => t.RequestPayload.Contains(reportId.ToString()))
                .FirstAsync();

            try
            {
                // 1. 更新任务状态为执行中
                task.TaskStatus = "EXECUTING";
                task.UpdateTime = DateTime.Now;
                await _db.Updateable(task).ExecuteCommandAsync();

                // 2. 查询报告数据
                var report = await _db.Queryable<RcConsultationReportEntity>()
                    .Where(r => r.Id == reportId)
                    .FirstAsync();

                var consultation = await _db.Queryable<RcConsultationEntity>()
                    .Where(c => c.Id == report.ConsultationId)
                    .FirstAsync();

                var patient = await _db.Queryable<RcPatientCaseEntity>()
                    .Where(p => p.Id == consultation.CaseId)
                    .FirstAsync();

                // 3. 调用EMR接口同步
                var emrData = new
                {
                    ReportId = reportId,
                    PatientId = patient.HisPatientId,
                    PatientName = patient.PatientName,
                    ConsultationTime = consultation.ScheduledStartTime,
                    Summary = report.Summary,
                    Diagnosis = report.Diagnosis,
                    TreatmentAdvice = report.TreatmentAdvice,
                    FollowUpPlan = report.FollowUpPlan,
                    SignTime = report.SignedTime
                };

                var syncResult = await _emrService.SyncReport(emrData);

                // 4. 更新任务状态
                task.TaskStatus = syncResult.Success ? "SUCCESS" : "FAILED";
                task.ResponsePayload = System.Text.Json.JsonSerializer.Serialize(new
                {
                    syncResult.Message,
                    syncResult.EMRReportId
                });
                task.CompletedTime = DateTime.Now;
                task.UpdateTime = DateTime.Now;

                await _db.Updateable(task).ExecuteCommandAsync();

                // 5. 发送同步完成通知
                if (syncResult.Success)
                {
                    await _eventBus.PublishAsync(new ReportSyncedEvent
                    {
                        ReportId = reportId,
                        EMRReportId = syncResult.EMRReportId
                    });
                }
            }
            catch (Exception ex)
            {
                // 记录错误并重试
                task.TaskStatus = "FAILED";
                task.ResponsePayload = System.Text.Json.JsonSerializer.Serialize(new
                {
                    Error = ex.Message,
                    StackTrace = ex.StackTrace
                });
                task.RetryCount++;
                task.UpdateTime = DateTime.Now;

                await _db.Updateable(task).ExecuteCommandAsync();

                // 如果重试次数未达到上限，重新入队
                if (task.RetryCount < 3)
                {
                    _backgroundJob.Schedule<ReportSyncJob>(
                        job => job.SyncToEMR(reportId),
                        TimeSpan.FromMinutes(5 * task.RetryCount));
                }

                throw;
            }
        }
    }

    // EMR集成服务接口
    public interface IEMRIntegrationService
    {
        Task<EMRSyncResult> SyncReport(object reportData);
    }

    public class EMRSyncResult
    {
        public bool Success { get; set; }
        public string Message { get; set; }
        public string EMRReportId { get; set; }
    }
}
```

```csharp
// PurestAdmin.Multiplex/EventHandlers/ReportEventHandler.cs
using System;
using System.Threading.Tasks;
using PurestAdmin.Multiplex.Contracts.Events;
using PurestAdmin.Zero.Notifications;

namespace PurestAdmin.Multiplex.EventHandlers
{
    public class ReportEventHandler :
        IEventHandler<ReportSubmittedEvent>,
        IEventHandler<ReportSignedEvent>,
        IEventHandler<ReportSigningCompletedEvent>,
        IEventHandler<ReportSyncedEvent>
    {
        private readonly INotificationService _notificationService;
        private readonly ISqlSugarClient _db;

        public ReportEventHandler(INotificationService notificationService, ISqlSugarClient db)
        {
            _notificationService = notificationService;
            _db = db;
        }

        /// <summary>
        /// 处理报告提交签署事件
        /// </summary>
        public async Task HandleAsync(ReportSubmittedEvent eventData)
        {
            // 发送签署通知给第一个签署人
            var firstSigner = eventData.SignFlows.OrderBy(f => f.SignOrder).First();
            
            await _notificationService.SendAsync(new NotificationMessage
            {
                UserId = firstSigner.SignerId,
                Title = "待签署报告",
                Content = $"您有一份会诊报告需要签署（报告ID：{eventData.ReportId}）",
                Type = "SIGN_REQUEST",
                Url = $"/consult/report-sign/{eventData.ReportId}"
            });

            // 发送短信通知（可选）
            // await _smsService.SendAsync(phone, "您有待签署的会诊报告，请及时处理");
        }

        /// <summary>
        /// 处理单个签署完成事件
        /// </summary>
        public async Task HandleAsync(ReportSignedEvent eventData)
        {
            // 查询下一个签署人
            var currentFlow = await _db.Queryable<RcReportSignFlowEntity>()
                .Where(f => f.Id == eventData.SignFlowId)
                .FirstAsync();

            var nextFlow = await _db.Queryable<RcReportSignFlowEntity>()
                .Where(f => f.ReportId == eventData.ReportId && f.SignOrder > currentFlow.SignOrder)
                .OrderBy(f => f.SignOrder)
                .FirstAsync();

            if (nextFlow != null)
            {
                // 通知下一个签署人
                await _notificationService.SendAsync(new NotificationMessage
                {
                    UserId = nextFlow.SignerId,
                    Title = "待签署报告",
                    Content = $"会诊报告已由{GetUserName(eventData.SignerId)}签署，请您继续签署（报告ID：{eventData.ReportId}）",
                    Type = "SIGN_REQUEST",
                    Url = $"/consult/report-sign/{eventData.ReportId}"
                });
            }

            // 记录签署日志
            await _db.Insertable(new RcConsultationTimelineEntity
            {
                ConsultationId = (await _db.Queryable<RcConsultationReportEntity>()
                    .Where(r => r.Id == eventData.ReportId)
                    .FirstAsync()).ConsultationId,
                EventCode = "REPORT_SIGNED",
                EventContent = $"报告由{GetUserName(eventData.SignerId)}签署",
                Snapshot = System.Text.Json.JsonSerializer.Serialize(new
                {
                    ReportId = eventData.ReportId,
                    SignerId = eventData.SignerId,
                    SignTime = DateTime.Now
                }),
                CreateBy = eventData.SignerId,
                CreateTime = DateTime.Now
            }).ExecuteCommandAsync();
        }

        /// <summary>
        /// 处理所有签署完成事件
        /// </summary>
        public async Task HandleAsync(ReportSigningCompletedEvent eventData)
        {
            // 通知所有会诊参与者
            var report = await _db.Queryable<RcConsultationReportEntity>()
                .Where(r => r.Id == eventData.ReportId)
                .FirstAsync();

            var members = await _db.Queryable<RcConsultationMemberEntity>()
                .Where(m => m.ConsultationId == report.ConsultationId)
                .ToListAsync();

            foreach (var member in members)
            {
                await _notificationService.SendAsync(new NotificationMessage
                {
                    UserId = member.UserId,
                    Title = "会诊报告签署完成",
                    Content = $"会诊报告已完成所有签署，即将归档（报告ID：{eventData.ReportId}）",
                    Type = "REPORT_COMPLETED",
                    Url = $"/consult/report-detail/{eventData.ReportId}"
                });
            }

            // 记录归档事件
            await _db.Insertable(new RcConsultationTimelineEntity
            {
                ConsultationId = report.ConsultationId,
                EventCode = "REPORT_SIGNING_COMPLETED",
                EventContent = "报告签署流程完成，等待归档",
                Snapshot = System.Text.Json.JsonSerializer.Serialize(new
                {
                    ReportId = eventData.ReportId,
                    CompleteTime = DateTime.Now
                }),
                CreateBy = CurrentUser.Id,
                CreateTime = DateTime.Now
            }).ExecuteCommandAsync();
        }

        /// <summary>
        /// 处理报告同步完成事件
        /// </summary>
        public async Task HandleAsync(ReportSyncedEvent eventData)
        {
            // 通知报告已归档到EMR
            var report = await _db.Queryable<RcConsultationReportEntity>()
                .Where(r => r.Id == eventData.ReportId)
                .FirstAsync();

            var members = await _db.Queryable<RcConsultationMemberEntity>()
                .Where(m => m.ConsultationId == report.ConsultationId)
                .ToListAsync();

            foreach (var member in members)
            {
                await _notificationService.SendAsync(new NotificationMessage
                {
                    UserId = member.UserId,
                    Title = "报告已归档",
                    Content = $"会诊报告已成功归档至EMR系统（EMR报告ID：{eventData.EMRReportId}）",
                    Type = "REPORT_ARCHIVED",
                    Url = $"/consult/report-detail/{eventData.ReportId}"
                });
            }

            // 记录归档完成事件
            await _db.Insertable(new RcConsultationTimelineEntity
            {
                ConsultationId = report.ConsultationId,
                EventCode = "REPORT_ARCHIVED",
                EventContent = $"报告已归档至EMR系统（ID：{eventData.EMRReportId}）",
                Snapshot = System.Text.Json.JsonSerializer.Serialize(new
                {
                    ReportId = eventData.ReportId,
                    EMRReportId = eventData.EMRReportId,
                    ArchiveTime = DateTime.Now
                }),
                CreateBy = CurrentUser.Id,
                CreateTime = DateTime.Now
            }).ExecuteCommandAsync();
        }
```
